יום ראשון
עמוד 24

[במוצאי מנוחה](https://he.wikipedia.org/wiki/%D7%91%D7%9E%D7%95%D7%A6%D7%90%D7%99_%D7%9E%D7%A0%D7%95%D7%97%D7%94_%D7%A7%D7%93%D7%9E%D7%A0%D7%95%D7%9A_%D7%AA%D7%97%D7%99%D7%9C%D7%94)

סימן: א"ב

בְּמוֹצָאֵי מְנוּחָה קִדַּמְנֽוּךָ תְּחִלָּה
הַט אׇזְנְךָ מִמָּרוֹם יוֹשֵׁב תְּהִלָּה
   לִשְׁמֹֽעַ אֶל הָרִנָּה וְאֶל הַתְּפִלָּה (מ"א ח כח) 

אֶת יְמִין עֹז עוֹרְרָה לַעֲשׂוֹת חַֽיִל
בְּצֶֽדֶק נֶעֱקַד וְנִשְׁחַט תְּמוּרוֹ אַֽיִל
גְּנֹן נָא גִזְעוֹ בְּזַעֲקָם [נוסח מערבי: בַּלָיִל נוסח מזרחי: בְּעוֹד לָֽיִל]

    לִשְׁמֹֽעַ אֶל הָרִנָּה וְאֶל הַתְּפִלָּה

דְּרוֹשׁ נָא דֹרְשֶֽׁיךָ בְּדׇרְשָׁם פָּנֶֽיךָ
הִדָּֽרֶשׁ לָֽמוֹ מִשְּׁמֵי מְעוֹנֶֽךָ
וּלְשַׁוְעַת [נוסח מערבי: מְחַנְּנֶֽיךָ נוסח מזרחי: חִנּוּנָם] אַל תַּעְלֵם אׇזְנֶֽךָ

    לִשְׁמֹֽעַ אֶל הָרִנָּה וְאֶל הַתְּפִלָּה

זוֹחֲלִים וְרוֹעֲדִים מִיּוֹם בּוֹאֶֽךָ
חָלִים כְּמַבְכִּירָה מֵעֶבְרַת מַשָּׂאֶֽךָ
טִנּוּפָם מְחֵה נָא [נוסח מערבי: וְיָרֽוֹנוּ פְלָאֶֽיךָ נוסח מזרחי: וְיוֹדוּ פִלְאֶֽךָ]

    לִשְׁמֹֽעַ אֶל הָרִנָּה וְאֶל הַתְּפִלָּה

יוֹצֵר אַתָּה לְכׇל יְצִיר נוֹצָר
כּוֹנַֽנְתָּ מֵאָז תֶּֽרֶף לְחַלְּצָם [נוסח מערבי: מִמַּעֲצָר נוסח מזרחי: מִמֵּצָר]
לְחׇנְנָם חִנָּם מֵאוֹצָר [נוסח מערבי: טוֹב] הַמְנֻצָּר

    לִשְׁמֹֽעַ אֶל הָרִנָּה וְאֶל הַתְּפִלָּה

מָרוֹם אִם עָצְמוּ פִּשְׁעֵי קְהָלֶֽךָ
נָא שַׂגְּבֵם מֵאוֹצָר הַמּוּכָן בִּזְבוּלֶֽךָ
[נוסח מערבי: עֵדֶֽיךָ נוסח מזרחי: עָדֶֽיךָ] לָחֹן חִנָּם בָּאִים אֵלֶֽיךָ

    לִשְׁמֹֽעַ אֶל הָרִנָּה וְאֶל הַתְּפִלָּה

פְּנֵה נָא אֶל הַתְּלָאוֹת וְאַל [נוסח מערבי: תֵֽפֶן] לַחֲטָאוֹת
צַדֵּק צוֹעֲקֶֽיךָ מַפְלִיא פְלָאוֹת
קְשֹׁב נָא [נוסח מערבי: רִנָּתָם נוסח מזרחי: חִנּוּנָם] [נוסח מערבי: אֱלֹהֵי הַצְּבָאוֹת נוסח מזרחי: אֱלֹהִים יְיָ צְבָאוֹת]

    לִשְׁמֹֽעַ אֶל הָרִנָּה וְאֶל הַתְּפִלָּה

רְצֵה עֲתִירָתָם בְּעׇמְדָם בַּלֵּילוֹת
שְׁעֵה נָא [נוסח מערבי: תְפִלָּתָם בְּ־ נוסח מזרחי: בְ]רָצוֹן [נוסח מערבי: כְּקׇרְבַּן תָּמִיד נוסח מזרחי: כְּקׇרְבָּן כָּלִיל] וְעוֹלוֹת
תַּרְאֵם נִסֶּֽיךָ עֹשֶׂה גְדֹלוֹת

    לִשְׁמֹֽעַ אֶל הָרִנָּה וְאֶל הַתְּפִלָּה
